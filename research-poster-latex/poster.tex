\Subhead{Nondimensionalization}
Scale concentrations by characteristic values and time by
$\beta$-catenin turnover, $d_B=1\,\mathrm{h}^{-1}$:
\Equation{1}{X=X_\ast x,\qquad \tau=d_Bt,\qquad
\varepsilon_X=\frac{d_B}{d_X}.}
\Caption{Hill activation: $H_n(u;\kappa)=u^n/(\kappa^n+u^n)$.
A dot below denotes $d/d\tau$.}
\par\vspace{9pt}
\Equation{2}{
\begin{aligned}
\dot b &=
 W+\eta_{13}H_{n_H}(h_{13};\kappa_{13})
 -b-\lambda_Ppb-\lambda_5\frac{h_5b}{\kappa_5+b},
 \\[4pt]
\varepsilon_P\dot p &=
 \frac{1+\rho_5h_5}{1+\rho_Bb+\rho_{13}h_{13}}-D_P(\theta_P)p,
 \\[4pt]
\varepsilon_5\dot h_5 &=
 a_5+\eta_RH_1(r;\kappa_R)-h_5-\eta_M\frac{mh_5}{\kappa_M+m},
 \\[4pt]
\varepsilon_{13}\dot h_{13} &=
 a_{13}+\eta_{B13}H_{n_B}(b;\kappa_{B13})
 +\eta_{M13}H_{n_M}(m;\kappa_{M13})-h_{13},
 \\[4pt]
\varepsilon_M\dot m &=
 a_M+\eta_{BM}H_{n_B}(b;\kappa_{BM})-m,
 \\[4pt]
\varepsilon_R\dot r &= \mu_R(\tau)-r-\lambda_Ccr,
 \\[4pt]
\varepsilon_C\dot c &=
 a_C+\eta_{RC}H_1(r;\kappa_{RC})+\eta_{BC}H_{n_B}(b;\kappa_{BC})-c .
\end{aligned}}

\Subhead{APC functioning and retinoid forcing}
\Equation{3}{\begin{aligned}
D_P(\theta_P)&=1+\delta_{P1}(1-\theta_P),\\[8pt]
\mu_R(\tau)&=\mu_0+A_R\!\left[1+\cos\!\left(\frac{2\pi\tau}{T_R}-\varphi\right)\right]\\[8pt]
&\quad+\frac{D_R}{2}
\left[\tanh q(\tau-\tau_1)-\tanh q(\tau-\tau_2)\right].
\end{aligned}}
\Caption{Reference: Radau on $\tau\in[0,150]$; ATRA pulse $[40,88]$.
Relative timescales $\varepsilon_R=0.40$, $\varepsilon_M=0.60$ produce mild stiffness.}

% ==================== 3. DIAGNOSTICS AND UDEs ====================
\sbox{\RightColumn}{\PanelStart{\RightWidth}
\SectionTitle{03}{Identifiability \& Bayesian uncertainty}
\Subhead{Fisher information and parameter recovery}
Local sensitivities at truth: 7 states, 10 conditions and 200 times.
\Equation{5}{\mathcal I=\sigma^{-2}J^{\mathsf T}J,\qquad
J_{ki}=\frac{\partial y_k}{\partial\log\theta_i},\qquad \sigma=0.002.}
\par\vspace{8pt}\noindent\hfill
\includegraphics[width=14.0in]{assets/fisher.pdf}\hfill\null
\Caption{\textbf{Fig. 5.} Severe-regime
$C_{ij}=\mathcal I_{ij}/\sqrt{\mathcal I_{ii}\mathcal I_{jj}}$:
$|C_{ij}|\approx1$ signals overlapping sensitivities.
Each regime has \textbf{one near-null direction} ($\lambda<1$); plotting floor $10^{-4}$.}

\Subhead{Bayesian parameter distributions}
Frozen-state HMC samples all 36 parameters using the integral likelihood.\textsuperscript{3}
\par\vspace{8pt}\includegraphics[width=\linewidth]{assets/presentation_marginals.pdf}
\Caption{\textbf{Fig. 6.} Presentation marginals (2000 draws/regime): narrow distributions can miss truth.}
\Caption{\textbf{Calibration limit.} HMC fails effective-sample-size and coverage
checks; interval widths are unreliable.}
