\Subhead{Nondimensionalization}
Scale concentrations by characteristic values and time by
$\beta$-catenin turnover, $d_B=1\,\mathrm{h}^{-1}$:
\Equation{1}{X=X_\ast x,\qquad \tau=d_Bt,\qquad
\varepsilon_X=\frac{d_B}{d_X}.}
\Caption{Hill activation: $H_n(u;\kappa)=u^n/(\kappa^n+u^n)$.
A dot below denotes $d/d\tau$.}
\par\vspace{9pt}
\Equation{2}{
\begin{aligned}
\dot b &=
 W+\eta_{13}H_{n_H}(h_{13};\kappa_{13})
 -b-\lambda_Ppb-\lambda_5\frac{h_5b}{\kappa_5+b},
 \\[4pt]
\varepsilon_P\dot p &=
 \frac{1+\rho_5h_5}{1+\rho_Bb+\rho_{13}h_{13}}-D_P(\theta_P)p,
 \\[4pt]
\varepsilon_5\dot h_5 &=
 a_5+\eta_RH_1(r;\kappa_R)-h_5-\eta_M\frac{mh_5}{\kappa_M+m},
 \\[4pt]
\varepsilon_{13}\dot h_{13} &=
 a_{13}+\eta_{B13}H_{n_B}(b;\kappa_{B13})
 +\eta_{M13}H_{n_M}(m;\kappa_{M13})-h_{13},
 \\[4pt]
\varepsilon_M\dot m &=
 a_M+\eta_{BM}H_{n_B}(b;\kappa_{BM})-m,
 \\[4pt]
\varepsilon_R\dot r &= \mu_R(\tau)-r-\lambda_Ccr,
 \\[4pt]
\varepsilon_C\dot c &=
 a_C+\eta_{RC}H_1(r;\kappa_{RC})+\eta_{BC}H_{n_B}(b;\kappa_{BC})-c .
\end{aligned}}

\Subhead{APC loss and retinoid forcing}
\Equation{3}{\begin{aligned}
D_P(\theta_P)&=1+\delta_{P1}(1-\theta_P),\\[8pt]
\mu_R(\tau)&=\mu_0+A_R\!\left[1+\cos\!\left(\frac{2\pi\tau}{T_R}-\varphi\right)\right]\\[8pt]
&\quad+\frac{D_R}{2}
\left[\tanh q(\tau-\tau_1)-\tanh q(\tau-\tau_2)\right].
\end{aligned}}
\Caption{Reference: Radau on $\tau\in[0,150]$; ATRA pulse $[40,88]$.
Relative timescales $\varepsilon_R=0.40$, $\varepsilon_M=0.60$ produce mild stiffness.}

\Note{\textbf{Derived readout: stemness}
\[
S(\tau)=\frac{b(1+\alpha_{13}h_{13})}{(1+p)(1+\alpha_5h_5)},
\qquad \alpha_5=\alpha_{13}=1.
\]
\CaptionFont Higher $S$ represents greater stemness in this model.}
\PanelEnd}

% ==================== 2. PINNs ====================
\sbox{\MiddleColumn}{\PanelStart{\MiddleWidth}
\SectionTitle{02}{PINNs \& identifiability}
\textbf{Forward problem.} Fourier features resolve circadian forcing
and the ATRA pulse.\textsuperscript{1}
\par\vspace{8pt}\noindent\input{assets/pinn-architecture.tex}\par
\Equation{4}{\mathcal L_{\rm phys}=\frac{1}{N_c}\sum_k
\left\|\partial_\tau\widehat{\mathbf z}_k-
\mathbf F(\tau_k,\widehat{\mathbf z}_k;\boldsymbol\theta)\right\|^2.}
\Caption{Data and initial-state losses are squared errors; physics derivatives use automatic differentiation.}
\par\vspace{10pt}\noindent\input{assets/forward-window.tex}\par
\Caption{\textbf{Fig. 2.} $\beta$-catenin and HOXA5, all four regimes.
Solid: Radau; dashed: PINN (40 observations).}
\Caption{\textbf{Forward error (mean relative $L^2$):}
\textbf{\ForwardDenseError\%} supervised / 100 labels;
\textbf{\ForwardSparseError\%} PINN / 40 labels.
Means span 7 states and 4 regimes; separate formulations.}

\Subhead{Recovering the 36 model parameters}
One state network per condition; 36 shared model parameters:
\Equation{5}{\mathbf r_k=
\widehat{\mathbf z}_{k+1}-\widehat{\mathbf z}_k
-\tfrac{1}{2}\Delta\tau_k\big(\mathbf F_k+\mathbf F_{k+1}\big).}
\Caption{Trapezoidal residual: no time derivatives; relative state weights; fixed collocation; two starts.}

% ==================== 3. NEURAL-MECHANISTIC HYBRID ====================
\sbox{\RightColumn}{\PanelStart{\RightWidth}
\SectionTitle{03}{Neural–mechanistic hybrid}
\textbf{Learn one relationship within the ODE model.}
A shared $2\times5$ tanh network replaces a regulatory term;
the remaining biology stays mechanistic.\textsuperscript{2}
\Equation{6}{\begin{aligned}
\eta_{BM}H_{n_B}(b;\kappa_{BM})&\longrightarrow f_\phi(b),\qquad f_\phi(0)=0,\\[9pt]
\varepsilon_M\dot m&=\textcolor{Garnet}{a_M}+
\textcolor{DataBlue}{f_\phi(b)}-m.
\end{aligned}}
\Caption{For MYC, $(\eta_{BM},\kappa_{BM})$ leave the inverse parameter set:
36 $\to$ 34. State networks, remaining parameters and $f_\phi$ are fitted
jointly with the integral residual; neural L2 penalty $10^{-8}$.}
\par\vspace{8pt}
\includegraphics[width=\linewidth]{assets/learned-myc.pdf}
\Caption{\textbf{Fig. 4.} Learned MYC activation versus the true mechanism,
in two regimes. Dots: enforced anchor; NRMSE uses observed ranges.}

\Subhead{Why an unvisited anchor can fail}
On the observed range, basal production and neural activation can compensate:
\Equation{7}{a_M+f_\phi(b)=(a_M+c)+\big[f_\phi(b)-c\big].}
\par\vspace{8pt}
\textbf{13.8–202.8\% basal error} across five constraint forms
in the Normal-regime screen.
